# NepalDisaster.com — National Emergency Information Service
# Public emergency information. Crawling is welcome and encouraged.

User-agent: *
Allow: /

# Server APIs are not public content. (The admin page is kept out of search by
# its X-Robots-Tag: noindex header in vercel.json — it is deliberately NOT named
# here, because robots.txt is public and would advertise where it lives.)
Disallow: /api/

# Favicons and styles must stay crawlable — Google needs to fetch the icon
# to show it beside the result, and CSS/JS to render the page for indexing.
Allow: /favicon/
Allow: /favicon.ico
Allow: /assets/

# Major crawlers: no crawl-delay, this is time-critical emergency information.
User-agent: Googlebot
Allow: /
Disallow: /api/

User-agent: Googlebot-Image
Allow: /

User-agent: Googlebot-News
Allow: /

User-agent: Bingbot
Allow: /
Disallow: /api/

Sitemap: https://www.nepaldisaster.com/sitemap.xml
